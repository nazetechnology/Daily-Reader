DAILY-PROXY.JS — NODE, ZERO DEPENDENCIES

Save this as daily-proxy.js, run node daily-proxy.js, and keep the proxy URL above set to http://localhost:8010. Done — the page detects it automatically. Tip: set the DAILY_KEY env var to keep the key on the server only.

// daily-proxy.js — daily.dev Public API CORS proxy (zero deps)
// run: node daily-proxy.js
// or:  DAILY_KEY=xxx node daily-proxy.js
const http = require('http');
const https = require('https');
const { URL } = require('url');

const PORT = Number(process.env.PORT) || 8010;
const KEY = process.env.DAILY_KEY || '';
const DAILY_HOST = 'api.daily.dev';
const DAILY_BASE = '/public/v1';

function sendJSON(res, status, data) {
  const body = JSON.stringify(data);
  res.writeHead(status, {
    'Content-Type': 'application/json; charset=utf-8',
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'Authorization, api-key, Content-Type',
    'Access-Control-Allow-Methods': 'GET, OPTIONS',
    'Cache-Control': 'no-store'
  });
  res.end(body);
}

function getToken(req) {
  if (KEY) return KEY;
  const auth = req.headers.authorization;
  if (auth && /^Bearer\s+/i.test(auth)) return auth.replace(/^Bearer\s+/i, '').trim();
  const apiKey = req.headers['api-key'];
  return apiKey ? String(apiKey).trim() : '';
}

function proxyRequest(req, res) {
  const incoming = new URL(req.url, `http://${req.headers.host || 'localhost'}`);
  const token = getToken(req);
  const headers = { Accept: 'application/json', 'User-Agent': 'Daily-Reader-Proxy/1.0' };
  if (token) headers.Authorization = 'Bearer ' + token;
  const target = DAILY_BASE + incoming.pathname + (incoming.search || '');

  console.log(`[${new Date().toLocaleTimeString()}] ${req.method} ${req.url} → https://${DAILY_HOST}${target}`);

  const up = https.request({ hostname: DAILY_HOST, port: 443, method: 'GET', path: target, headers }, upstream => {
    let body = '';
    upstream.setEncoding('utf8');
    upstream.on('data', chunk => body += chunk);
    upstream.on('end', () => {
      let json;
      try { json = JSON.parse(body); }
      catch (_) { return sendJSON(res, upstream.statusCode || 502, { error: 'invalid_upstream_response', message: 'daily.dev returned a non-JSON response', upstreamStatus: upstream.statusCode }); }
      sendJSON(res, upstream.statusCode || 502, json);
    });
  });
  up.on('error', err => sendJSON(res, 502, { error: 'upstream_failed', message: err.message }));
  up.end();
}

http.createServer((req, res) => {
  if (req.method === 'OPTIONS') {
    res.writeHead(204, { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Headers': 'Authorization, api-key, Content-Type', 'Access-Control-Allow-Methods': 'GET, OPTIONS' });
    return res.end();
  }
  if (req.method !== 'GET') return sendJSON(res, 405, { error: 'method_not_allowed' });
  try { proxyRequest(req, res); } catch (err) { sendJSON(res, 500, { error: 'proxy_error', message: err.message }); }
}).listen(PORT, () => console.log(`daily.dev proxy → http://localhost:${PORT}`));
